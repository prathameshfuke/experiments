// 1) Start Microwind, File > Select Foundry > cmos012.RUL.
// 2) Compile > Compile Verilog File, choose this HALFADDER.TXT, click Compile (layout is generated).

module hadd(a, b, sum, carry);
 input a, b;
 output sum, carry;
 xor xor1(sum, a, b);
 and and1(carry, a, b);
endmodule
